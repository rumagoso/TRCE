        function generatePrompt() {
            const categoryFilter = document.getElementById('categorySelect').value;
            const mode = document.getElementById('modeSelect').value;
            
            // Filter database
            let filtered = database;
            if (categoryFilter !== 'all') {
                filtered = database.filter(item => item.catId === categoryFilter);
            }

            // Pick a random item
            const randomIndex = Math.floor(Math.random() * filtered.length);
            const selected = filtered[randomIndex];

            // Get elements
            const card = document.getElementById('card');
            const catDisplay = document.getElementById('catDisplay');
            const modeDisplay = document.getElementById('modeDisplay');
            const promptDisplay = document.getElementById('promptDisplay');

            // Trigger animation
            card.classList.remove('fade-in');
            void card.offsetWidth; // trigger reflow
            card.classList.add('fade-in');

            // Update content
            catDisplay.textContent = selected.catName;
            modeDisplay.textContent = mode.toUpperCase();
            promptDisplay.textContent = mode === 'writing' ? selected.writing : selected.drawing;
        }
